#pragma once

namespace MathFunc {
	int FactoN(int N) {
		int fac = 1;
		for (int i = N;i >= 1;i--) {
			fac = fac * i;
		}
		return fac;
	}
	int MaxOf2Nums(int Num1, int Num2) {
		if (Num1 > Num2)
			return Num1;
		else
			return Num2;
	}
	void MySwapNumsProducer(int& a, int& b) {
		int x;
		x = a;
		a = b;
		b = x;
	}
	int SumOfArray(int arr[100], int arrLengh) {
		int Sum = 0;
		for (int i = 0;i < arrLengh;i++) {
			Sum += arr[i];
		}
		return Sum;
	}


}
